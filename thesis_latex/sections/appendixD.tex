\section{Derivation of the Trajectory Bound}
\label{app:detailed_proof}

This appendix derives the two results used in Section~\ref{subsec:reference}: the replay loss is monotone along the reference path (Proposition~\ref{prop:monotone}), and a gradient-flow iterate under a linear ramp tracks that path with $O(1/N)$ lag (Proposition~\ref{prop:tracking}), which together give the transient bound of Equation~\eqref{eq:bound}.

\paragraph{Setup.}
For $\lambda \in [0,1]$ define the curriculum objective
\begin{equation}
    L_\lambda(\theta) \;=\; L_{\text{replay}}(\theta) + \lambda\, L_{\text{current}}(\theta),
    \qquad
    H_\lambda(\theta) \;=\; H_{\text{replay}}(\theta) + \lambda\, H_{\text{current}}(\theta),
\end{equation}
with $H_{\text{replay}} = \nabla^2 L_{\text{replay}}$ and $H_{\text{current}} = \nabla^2 L_{\text{current}}$. We make two assumptions.

\begin{itemize}
    \item[\textbf{(A1)}] $L_{\text{replay}}$ and $L_{\text{current}}$ are $C^2$ in a neighbourhood of the path considered below.
    \item[\textbf{(A2)}] There is a $C^1$ branch of critical points $\lambda \mapsto \Theta^*(\lambda)$ of $L_\lambda$ with $\Theta^*(0) = \theta_0^*$ and $H_\lambda(\Theta^*(\lambda)) \succ 0$ for all $\lambda \in [0,1]$.
\end{itemize}

Assumption (A2) asks for a tracked branch of strict local minima, not global convexity. Near $\lambda = 0$ it holds automatically: $\theta_0^*$ is a non-degenerate local minimum of $L_{\text{replay}}$, so $H_0(\theta_0^*) \succ 0$ and the implicit function theorem yields a unique $C^1$ branch locally. Its persistence is controlled by Weyl's inequality, $\mu_{\min}(H_\lambda) \ge \mu_{\min}(H_{\text{replay}}) + \lambda\, \mu_{\min}(H_{\text{current}})$, with $\mu_{\min}$ the smallest eigenvalue: positive-definiteness survives until the new task's negative curvature outweighs the old task's, and holds for all $\lambda$ whenever $H_{\text{current}} \succeq 0$ along the branch. On a deep non-convex backbone (A2) is a local modelling assumption, which is why Section~\ref{subsec:reference} presents the bound as motivation rather than a guarantee.

\begin{proposition}[Monotonicity of the replay loss]
    \label{prop:monotone}
    Under (A1)--(A2),
    \begin{equation*}
        \frac{d}{d\lambda}\, L_{\text{replay}}(\Theta^*(\lambda))
        \;=\; \lambda\, \nabla L_{\text{current}}^\top H_\lambda^{-1}\, \nabla L_{\text{current}}
        \;\ge\; 0
        \qquad \text{for all } \lambda \in [0,1],
    \end{equation*}
    where $\nabla L_{\text{current}}$ and $H_\lambda$ are evaluated at $\Theta^*(\lambda)$. In particular $L_{\text{replay}}(\Theta^*(\lambda)) \le L_{\text{replay}}(\theta_{\text{joint}}^*)$ for all $\lambda$, with $\theta_{\text{joint}}^* = \Theta^*(1)$.
\end{proposition}

\begin{proof}
    The branch satisfies the first-order condition
    \begin{equation}
        \nabla L_{\text{replay}}(\Theta^*(\lambda)) + \lambda\, \nabla L_{\text{current}}(\Theta^*(\lambda)) = 0.
        \label{eq:foc_detailed}
    \end{equation}
    Differentiating \eqref{eq:foc_detailed} in $\lambda$ and collecting terms gives
    $H_\lambda\, \tfrac{d\Theta^*}{d\lambda} + \nabla L_{\text{current}} = 0$, hence
    \begin{equation}
        \frac{d\Theta^*}{d\lambda} = -\,H_\lambda^{-1}\, \nabla L_{\text{current}}.
        \label{eq:path_movement}
    \end{equation}
    By the chain rule and \eqref{eq:foc_detailed},
    \begin{equation*}
        \frac{d}{d\lambda}\, L_{\text{replay}}(\Theta^*(\lambda))
        = \nabla L_{\text{replay}}^\top \frac{d\Theta^*}{d\lambda}
        = \left(-\lambda\, \nabla L_{\text{current}}\right)^\top \left(-H_\lambda^{-1}\, \nabla L_{\text{current}}\right)
        = \lambda\, \nabla L_{\text{current}}^\top H_\lambda^{-1}\, \nabla L_{\text{current}},
    \end{equation*}
    which is non-negative because $\lambda \ge 0$ and $H_\lambda^{-1} \succ 0$ by (A2). Integrating from $\lambda$ to $1$ gives the final statement.
\end{proof}

\begin{proposition}[$O(1/N)$ tracking lag]
    \label{prop:tracking}
    Let $\theta(t)$ follow the gradient flow $\dot\theta = -\nabla L_{\lambda(t)}(\theta)$ under the linear ramp $\lambda(t) = t/N$, with $\theta(0) = \theta_0^*$, and let $e(t) = \theta(t) - \Theta^*(\lambda(t))$ denote the tracking error. Suppose, in addition to (A1)--(A2), that along the branch $\mu_{\min}(H_\lambda) \ge \mu > 0$ and $\|\nabla L_{\text{current}}\| \le g$. Then, to first order in $e$, the error settles at
    \begin{equation*}
        \|e\| \;\le\; \frac{1}{N}\, \big\|H_\lambda^{-1}\big\|\, \Big\|\frac{d\Theta^*}{d\lambda}\Big\|
        \;\le\; \frac{g}{\mu^2 N}
        \;=\; O\!\left(\frac{1}{N}\right).
    \end{equation*}
\end{proposition}

\begin{proof}
    Differentiating the error and inserting the flow and the chain rule for the moving target,
    \begin{equation*}
        \dot e = -\nabla L_{\lambda(t)}(\theta(t)) - \frac{d\Theta^*}{d\lambda}\,\dot\lambda,
        \qquad \dot\lambda = \frac{1}{N}.
    \end{equation*}
    Linearising the gradient about the minimiser, $\nabla L_\lambda(\theta) = H_\lambda\, e + O(\|e\|^2)$ since $\nabla L_\lambda(\Theta^*) = 0$, yields the error dynamics
    \begin{equation*}
        \dot e \;\approx\; -H_\lambda\, e \;-\; \frac{1}{N}\, \frac{d\Theta^*}{d\lambda}.
    \end{equation*}
    Because $H_\lambda \succeq \mu I$, these dynamics contract toward the quasi-static equilibrium $e^\dagger = -\tfrac{1}{N} H_\lambda^{-1} \tfrac{d\Theta^*}{d\lambda}$; bounding $\|H_\lambda^{-1}\| \le 1/\mu$ and, by \eqref{eq:path_movement}, $\|d\Theta^*/d\lambda\| \le g/\mu$ gives the stated bound. The argument is a first-order, continuous-time approximation: it captures how the lag scales with the ramp length $N$, not an exact discrete-time constant.
\end{proof}

\paragraph{The transient bound.}
Expanding the replay loss about the reference point, $L_{\text{replay}}(\theta(t)) = L_{\text{replay}}(\Theta^*) + \nabla L_{\text{replay}}(\Theta^*)^\top e + O(\|e\|^2)$, Proposition~\ref{prop:monotone} bounds the first term by $L_{\text{replay}}(\theta_{\text{joint}}^*)$ and Proposition~\ref{prop:tracking} bounds the correction by $O(1/N)$ (the gradient norm $\|\nabla L_{\text{replay}}(\Theta^*)\| = \lambda \|\nabla L_{\text{current}}\| \le g$ is bounded along the branch). Hence
\begin{equation}
    \max_t\, L_{\text{replay}}(\theta(t)) \;\le\; L_{\text{replay}}(\theta_{\text{joint}}^*) + O\!\left(\frac{1}{N}\right):
\end{equation}
the permanent price is the first term, and the transient contribution of the tracking lag shrinks as the ramp lengthens, which is Equation~\eqref{eq:bound}.
